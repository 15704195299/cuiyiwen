智能客服系统请求处理流程

mermaid.js · flowchart TD
flowchart TD
    A([用户发起提问]) --> B{身份识别}
    B -->|已登录| C[获取用户画像]
    B -->|游客| D[匿名模式]
    C --> E[NLU 自然语言理解]
    D --> E
    E --> F{意图识别}
    F -->|FAQ 查询| G[知识库向量检索]
    F -->|闲聊| H[LLM 自由对话]
    F -->|业务办理| I[对话状态追踪 DST]
    G --> J{RAG 结果相关?}
    J -->|是| K[返回检索答案 + 引用]
    J -->|否| L[转人工客服]
    H --> M[生成回复]
    I --> N{多轮信息完整?}
    N -->|否| O[追问用户补充信息]
    N -->|是| P[调用后端 API 办理业务]
    K --> Q[输出回复]
    L --> Q
    M --> Q
    P --> Q
    O --> E
    Q --> R[记录对话日志]
    R --> S([结束])

    USER ||--o{ SESSION : "发起"

    INTENT ||--o{ MESSAGE : "识别为"
    FEEDBACK }o--|| MESSAGE : "评价"

    TRANSFER }o--|| SESSION : "转人工"

    USER {
        bigint user_id PK
        varchar username
        varchar email
        tinyint gender
        date birthday
        timestamp created_at
    }
    SESSION {
        bigint session_id PK
        bigint user_id FK
        tinyint channel
        tinyint status
        timestamp start_time
        timestamp end_time
        float satisfaction
    }
    CONTEXT {
        bigint context_id PK
        bigint session_id FK
        varchar current_intent
        json slot_values
        tinyint turn_count
    }
    MESSAGE {
        bigint msg_id PK
        bigint session_id FK
        tinyint role
        text content
        bigint intent_id FK
        vector embedding
        timestamp created_at
    }
    INTENT {
        bigint intent_id PK
        varchar intent_name
        tinyint domain
        text description
    }
    KNOWLEDGE {
        bigint doc_id PK
        varchar title
        text content
        vector embedding
        int source_type
        timestamp updated_at
    }
    TAG {
        bigint tag_id PK
        varchar tag_name
        varchar tag_color
    }
    KNOWLEDGE_TAG {
        bigint doc_id FK
        bigint tag_id FK
    }
    FEEDBACK {
        bigint feedback_id PK
        bigint user_id FK
        bigint msg_id FK
        tinyint rating
        text comment
        timestamp created_at
    }
    HUMAN_AGENT {
        bigint agent_id PK
        varchar agent_name
        tinyint status
        int current_load
    }
    TRANSFER {
        bigint transfer_id PK
        bigint session_id FK
        bigint agent_id FK
        tinyint reason
        timestamp transfer_time
    }
